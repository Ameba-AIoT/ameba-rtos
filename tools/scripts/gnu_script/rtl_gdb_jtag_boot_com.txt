# GDB script for loading ram.bin process
# Common script for all Ameba SOC platforms
# Platform-specific variables should be set via -ex before sourcing this script
# By gdb -ex set: GDB_PORT

#===============================================================================
# GDB connection
set remotetimeout 100000
# HOST_IP is set via gdb.py --remote-server (passed through -ex).
# Port $GDB_PORT will be set via -ex parameter before sourcing this script
# target remote :2335
printf "HOST_IP=[%s] GDB_PORT=[%d]\n", $HOST_IP, $GDB_PORT
eval "target remote %s:%d", $HOST_IP, $GDB_PORT

#===============================================================================
#Message display setting
#disable all messages

set verbose off
set complaints 0
set confirm off
set exec-done-display off
show exec-done-display
set trace-commands off
#set debug aix-thread off
#set debug dwarf2-die 0
set debug displaced off
set debug expression 0
set debug frame 0
set debug infrun 0
set debug observer 0
set debug overload 0
#set debugvarobj 0
set pagination off
set print address off
set print symbol-filename off
set print symbol off
set print pretty off
set print object off
#set debug notification off
set debug parser off
set debug remote 0

#===============================================================================
#Halt only, the caller script resets the cpu when needed (load rom / flash write)
monitor halt
monitor waithalt 2000

#===============================================================================
# Warm reset (monitor reset 0) does not reset peripherals: if crypto (AES/SHA/PKE) held its HW mutex
# when the cpu was reset, ROM CRYPTO_Init + pke_engine_init can never take the mutex again.
# Call this right after reset while the cpu is halted: clear crypto function/clock enable to reset the IP,
# ROM CRYPTO_Init enables it again from a clean state.
# By gdb -ex set: CRYPTO_RST_EN, and if enabled SYSTEM_CTRL_BASE, REG_LSYS_FEN_GRP0, REG_LSYS_CKE_GRP0, CRYPTO_RCC_MASK
define crypto_rcc_clear
    if ($CRYPTO_RST_EN == 1)
        set {int}($SYSTEM_CTRL_BASE + $REG_LSYS_FEN_GRP0) = {int}($SYSTEM_CTRL_BASE + $REG_LSYS_FEN_GRP0) & ~$CRYPTO_RCC_MASK
        set {int}($SYSTEM_CTRL_BASE + $REG_LSYS_CKE_GRP0) = {int}($SYSTEM_CTRL_BASE + $REG_LSYS_CKE_GRP0) & ~$CRYPTO_RCC_MASK
        printf "Crypto RCC clear, FEN_GRP0=0x%x, CKE_GRP0=0x%x\n", {int}($SYSTEM_CTRL_BASE + $REG_LSYS_FEN_GRP0), {int}($SYSTEM_CTRL_BASE + $REG_LSYS_CKE_GRP0)
    end
end
